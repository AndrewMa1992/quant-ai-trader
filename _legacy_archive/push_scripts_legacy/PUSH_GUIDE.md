# Quant AI Trader - GitHub 推送指南
# 打开 PowerShell（不是 WorkBuddy），逐行执行：

# 1. 进入项目目录
cd G:\workbuddyspace\quant-ai-trader

# 2. 添加远程仓库
C:\"Program Files"\Git\bin\git.exe remote add origin https://github.com/AndrewMa1992/quant-ai-trader.git

# 3. 改默认分支名
C:\"Program Files"\Git\bin\git.exe branch -m main

# 4. 推送（会弹出浏览器登录 GitHub）
C:\"Program Files"\Git\bin\git.exe push -u origin main --force
